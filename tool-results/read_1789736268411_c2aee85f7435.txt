   196→### 3. `src/trading_strategies/data/fetcher.py`
   197→
   198→```python
   199→"""Data fetching from yfinance with caching."""
   200→
   201→import pandas as pd
   202→import yfinance as yf
   203→from pathlib import Path
   204→from datetime import datetime, timedelta
   205→import logging
   206→from typing import List, Optional, Tuple
   207→
   208→logger = logging.getLogger(__name__)
   209→
   210→class DataFetcher:
   211→    """Fetch and cache financial data from yfinance."""
   212→    
   213→    def __init__(self, cache_dir: Path, cache_enabled: bool = True):
   214→        self.cache_dir = cache_dir
   215→        self.cache_enabled = cache_enabled
   216→        
   217→    def _get_cache_path(self, ticker: str, start_date: str, end_date: str) -> Path:
   218→        """Generate cache file path."""
   219→        filename = f"{ticker}_{start_date}_{end_date}.parquet"
   220→        return self.cache_dir / filename
   221→    
   222→    def _is_cache_valid(self, cache_path: Path, max_age_days: int = 1) -> bool:
   223→        """Check if cached data is still valid."""
   224→        if not cache_path.exists():
   225→            return False
   226→        
   227→        file_age = datetime.now() - datetime.fromtimestamp(cache_path.stat().st_mtime)
   228→        return file_age < timedelta(days=max_age_days)
   229→    
   230→    def fetch_ohlcv(
   231→        self,
   232→        ticker: str,
   233→        start_date: str,
   234→        end_date: str,
   235→        interval: str = "1d"
   236→    ) -> pd.DataFrame:
   237→        """
   238→        Fetch OHLCV data for a ticker.
   239→        
   240→        Args:
   241→            ticker: Stock ticker (e.g., 'AAPL')
   242→            start_date: Start date (YYYY-MM-DD)
   243→            end_date: End date (YYYY-MM-DD)
   244→            interval: Data interval ('1d', '1h', '5m', etc.)
   245→            
   246→        Returns:
   247→            DataFrame with OHLCV data
   248→        """
   249→        cache_path = self._get_cache_path(ticker, start_date, end_date)
   250→        
   251→        # Try to load from cache
   252→        if self.cache_enabled and self._is_cache_valid(cache_path):
   253→            logger.info(f"Loading {ticker} from cache")
   254→            return pd.read_parquet(cache_path)
   255→        
   256→        # Fetch from yfinance
   257→        logger.info(f"Fetching {ticker} from yfinance ({start_date} to {end_date})")
   258→        df = yf.download(
   259→            ticker,
   260→            start=start_date,
   261→            end=end_date,
   262→            interval=interval,
   263→            progress=False
   264→        )
   265→        
   266→        # Clean up column names
   267→        df.columns = [col.lower() for col in df.columns]
   268→        df.index.name = 'date'
   269→        
   270→        # Cache if enabled
   271→        if self.cache_enabled:
   272→            df.to_parquet(cache_path)
   273→            logger.info(f"Cached {ticker} to {cache_path}")
   274→        
   275→        return df
   276→    
   277→    def fetch_multiple(
   278→        self,
   279→        tickers: List[str],
   280→        start_date: str,
   281→        end_date: str
   282→    ) -> pd.DataFrame:
   283→        """
   284→        Fetch data for multiple tickers.
   285→        
   286→        Returns:
   287→            DataFrame with multi-level columns (Ticker, OHLCV)
   288→        """
   289→        data = {}
   290→        for ticker in tickers:
   291→            try:
   292→                data[ticker] = self.fetch_ohlcv(ticker, start_date, end_date)
   293→            except Exception as e:
   294→                logger.error(f"Failed to fetch {ticker}: {e}")
   295→        
   296→        # Combine into multi-level dataframe
   297→        return pd.concat(data, axis=1, keys=data.keys())
   298→    
   299→    def get_close_prices(
   300→        self,
   301→        tickers: List[str],
   302→        start_date: str,
   303→        end_date: str
   304→    ) -> pd.DataFrame:
   305→        """Get closing prices for multiple tickers."""
   306→        df = self.fetch_multiple(tickers, start_date, end_date)
   307→        return df.xs('close', level=1, axis=1)
   308→```

   312→### 4. `src/trading_strategies/strategies/base.py`
   313→
   314→```python
   315→"""Base strategy class for all strategies."""
   316→
   317→from abc import ABC, abstractmethod
   318→import pandas as pd
   319→import numpy as np
   320→from typing import Tuple, Dict, Any
   321→import logging
   322→
   323→logger = logging.getLogger(__name__)
   324→
   325→class BaseStrategy(ABC):
   326→    """
   327→    Abstract base class for all trading strategies.
   328→    
   329→    A strategy generates trading signals based on market data.
   330→    """
   331→    
   332→    def __init__(self, name: str, parameters: Dict[str, Any] = None):
   333→        """
   334→        Initialize strategy.
   335→        
   336→        Args:
   337→            name: Strategy name
   338→            parameters: Strategy parameters
   339→        """
   340→        self.name = name
   341→        self.parameters = parameters or {}
   342→        
   343→    @abstractmethod
   344→    def generate_signals(self, data: pd.DataFrame) -> pd.DataFrame:
   345→        """
   346→        Generate trading signals.
   347→        
   348→        Args:
   349→            data: DataFrame with OHLCV data
   350→            
   351→        Returns:
   352→            DataFrame with 'signal' column (1=long, -1=short, 0=neutral)
   353→        """
   354→        pass
   355→    
   356→    def calculate_returns(self, data: pd.DataFrame) -> pd.Series:
   357→        """Calculate daily returns."""
   358→        return data['close'].pct_change()
   359→    
   360→    def backtest(
   361→        self,
   362→        data: pd.DataFrame,
   363→        initial_capital: float = 100000,
   364→        commission: float = 0.0005
   365→    ) -> Dict[str, Any]:
   366→        """
   367→        Simple backtest of strategy.
   368→        
   369→        Args:
   370→            data: DataFrame with OHLCV data
   371→            initial_capital: Starting capital
   372→            commission: Commission per trade (as decimal)
   373→            
   374→        Returns:
   375→            Dictionary with backtest results
   376→        """
   377→        # Generate signals
   378→        signals_df = self.generate_signals(data.copy())
   379→        signals_df['signal'] = signals_df['signal'].fillna(0)
   380→        
   381→        # Calculate returns
   382→        returns = data['close'].pct_change()
   383→        
   384→        # Apply position from signals
   385→        signals_df['position'] = signals_df['signal'].shift(1).fillna(0)
   386→        
   387→        # Strategy returns
   388→        signals_df['strategy_return'] = signals_df['position'] * returns
   389→        
   390→        # Account for commissions
   391→        signals_df['trades'] = signals_df['position'].diff().abs()
   392→        signals_df['commission_cost'] = signals_df['trades'] * commission
   393→        signals_df['net_return'] = signals_df['strategy_return'] - signals_df['commission_cost']
   394→        
   395→        # Cumulative returns
   396→        signals_df['cum_return'] = (1 + signals_df['net_return']).cumprod()
   397→        signals_df['equity'] = initial_capital * signals_df['cum_return']
   398→        
   399→        # Calculate metrics
   400→        total_return = signals_df['cum_return'].iloc[-1] - 1
   401→        num_trades = (signals_df['trades'] > 0).sum()
   402→        
   403→        # Sharpe ratio (annualized)
   404→        sharpe = np.sqrt(252) * signals_df['net_return'].mean() / signals_df['net_return'].std()
   405→        
   406→        # Max drawdown
   407→        cum_returns = signals_df['cum_return']
   408→        running_max = cum_returns.expanding().max()
   409→        drawdown = (cum_returns - running_max) / running_max
   410→        max_drawdown = drawdown.min()
   411→        
   412→        results = {
   413→            'strategy': self.name,
   414→            'total_return': total_return,
   415→            'num_trades': num_trades,
   416→            'sharpe_ratio': sharpe,
   417→            'max_drawdown': max_drawdown,
   418→            'signals_df': signals_df,
   419→        }
   420→        
   421→        return results
   422→```

   426→### 5. `src/trading_strategies/strategies/stocks/moving_average.py`
   427→
   428→```python
   429→"""Moving Average strategy from paper (Section 3.11-3.13)."""
   430→
   431→import pandas as pd
   432→import numpy as np
   433→from typing import Dict, Any
   434→from ..base import BaseStrategy
   435→
   436→class SingleMovingAverageStrategy(BaseStrategy):
   437→    """
   438→    Strategy: Single Moving Average (from paper Section 3.11)
   439→    
   440→    Generate buy signal when price crosses above MA.
   441→    Generate sell signal when price crosses below MA.
   442→    """
   443→    
   444→    def __init__(self, ma_period: int = 50):
   445→        super().__init__(
   446→            name=f"SMA_{ma_period}",
   447→            parameters={'ma_period': ma_period}
   448→        )
   449→        self.ma_period = ma_period
   450→    
   451→    def generate_signals(self, data: pd.DataFrame) -> pd.DataFrame:
   452→        """
   453→        Generate trading signals based on price crossing MA.
   454→        
   455→        Signal logic:
   456→        - 1 (long): price > SMA
   457→        - -1 (short): price < SMA
   458→        - 0 (neutral): price ≈ SMA
   459→        """
   460→        df = data.copy()
   461→        df['sma'] = df['close'].rolling(window=self.ma_period).mean()
   462→        
   463→        # Signal: 1 if price > SMA, -1 if price < SMA
   464→        df['signal'] = np.where(df['close'] > df['sma'], 1, -1)
   465→        
   466→        # Neutral signal until we have enough data
   467→        df.loc[:self.ma_period, 'signal'] = 0
   468→        
   469→        return df[['signal']]
   470→
   471→
   472→class DualMovingAverageStrategy(BaseStrategy):
   473→    """
   474→    Strategy: Dual Moving Average (from paper Section 3.12)
   475→    
   476→    Buy signal when fast MA crosses above slow MA.
   477→    Sell signal when fast MA crosses below slow MA.
   478→    """
   479→    
   480→    def __init__(self, fast_period: int = 20, slow_period: int = 50):
   481→        super().__init__(
   482→            name=f"DMA_{fast_period}_{slow_period}",
   483→            parameters={'fast_period': fast_period, 'slow_period': slow_period}
   484→        )
   485→        self.fast_period = fast_period
   486→        self.slow_period = slow_period
   487→    
   488→    def generate_signals(self, data: pd.DataFrame) -> pd.DataFrame:
   489→        """
   490→        Generate signals based on MA crossover.
   491→        
   492→        Signal logic:
   493→        - 1 (long): fast MA > slow MA
   494→        - -1 (short): fast MA < slow MA
   495→        """
   496→        df = data.copy()
   497→        df['fast_ma'] = df['close'].rolling(window=self.fast_period).mean()
   498→        df['slow_ma'] = df['close'].rolling(window=self.slow_period).mean()
   499→        
   500→        # Signal on crossover
   501→        df['signal'] = np.where(df['fast_ma'] > df['slow_ma'], 1, -1)
   502→        
   503→        # Neutral signal until we have enough data
   504→        df.loc[:self.slow_period, 'signal'] = 0
   505→        
   506→        return df[['signal']]
   507→
   508→
   509→class TripleMovingAverageStrategy(BaseStrategy):
   510→    """
   511→    Strategy: Triple Moving Average (from paper Section 3.13)
   512→    
   513→    Advanced trend confirmation using 3 MAs.
   514→    Buy when all MAs are in correct order (fast > medium > slow)
   515→    """
   516→    
   517→    def __init__(self, fast: int = 10, medium: int = 20, slow: int = 50):
   518→        super().__init__(
   519→            name=f"TMA_{fast}_{medium}_{slow}",
   520→            parameters={'fast': fast, 'medium': medium, 'slow': slow}
   521→        )
   522→        self.fast = fast
   523→        self.medium = medium
   524→        self.slow = slow
   525→    
   526→    def generate_signals(self, data: pd.DataFrame) -> pd.DataFrame:
   527→        """
   528→        Generate signals from triple MA configuration.
   529→        
   530→        Long: fast_MA > medium_MA > slow_MA
   531→        Short: fast_MA < medium_MA < slow_MA
   532→        Neutral: otherwise
   533→        """
   534→        df = data.copy()
   535→        df['fast_ma'] = df['close'].rolling(window=self.fast).mean()
   536→        df['medium_ma'] = df['close'].rolling(window=self.medium).mean()
   537→        df['slow_ma'] = df['close'].rolling(window=self.slow).mean()
   538→        
   539→        # Bullish: fast > medium > slow
   540→        bullish = (df['fast_ma'] > df['medium_ma']) & (df['medium_ma'] > df['slow_ma'])
   541→        # Bearish: fast < medium < slow
   542→        bearish = (df['fast_ma'] < df['medium_ma']) & (df['medium_ma'] < df['slow_ma'])
   543→        
   544→        df['signal'] = 0
   545→        df.loc[bullish, 'signal'] = 1
   546→        df.loc[bearish, 'signal'] = -1
   547→        
   548→        # Neutral until we have enough data
   549→        df.loc[:self.slow, 'signal'] = 0
   550→        
   551→        return df[['signal']]
   552→```

   556→### 6. `src/trading_strategies/strategies/stocks/mean_reversion.py`
   557→
   558→```python
   559→"""Mean Reversion strategy from paper (Section 3.9)."""
   560→
   561→import pandas as pd
   562→import numpy as np
   563→from typing import Tuple
   564→from ..base import BaseStrategy
   565→
   566→class MeanReversionStrategy(BaseStrategy):
   567→    """
   568→    Strategy: Mean Reversion (from paper Section 3.9)
   569→    
   570→    Assumes prices revert to their mean.
   571→    Generate signals based on deviation from moving average.
   572→    """
   573→    
   574→    def __init__(self, lookback: int = 20, z_score_threshold: float = 2.0):
   575→        super().__init__(
   576→            name=f"MeanReversion_{lookback}_{z_score_threshold}",
   577→            parameters={'lookback': lookback, 'z_score_threshold': z_score_threshold}
   578→        )
   579→        self.lookback = lookback
   580→        self.z_score_threshold = z_score_threshold
   581→    
   582→    def generate_signals(self, data: pd.DataFrame) -> pd.DataFrame:
   583→        """
   584→        Generate signals based on z-score from mean.
   585→        
   586→        Signal logic:
   587→        - 1 (long): price is z_score_threshold std below mean (buy dip)
   588→        - -1 (short): price is z_score_threshold std above mean (short spike)
   589→        - 0 (neutral): within threshold
   590→        """
   591→        df = data.copy()
   592→        
   593→        # Calculate rolling mean and std
   594→        df['mean'] = df['close'].rolling(window=self.lookback).mean()
   595→        df['std'] = df['close'].rolling(window=self.lookback).std()
   596→        
   597→        # Calculate z-score
   598→        df['z_score'] = (df['close'] - df['mean']) / df['std']
   599→        
   600→        # Generate signals
   601→        df['signal'] = 0
   602→        df.loc[df['z_score'] < -self.z_score_threshold, 'signal'] = 1   # Buy
   603→        df.loc[df['z_score'] > self.z_score_threshold, 'signal'] = -1    # Sell
   604→        
   605→        # Neutral signal until we have enough data
   606→        df.loc[:self.lookback, 'signal'] = 0
   607→        
   608→        return df[['signal']]
   609→
   610→
   611→class PairsStrategy(BaseStrategy):
   612→    """
   613→    Strategy: Pairs Trading (from paper Section 3.8)
   614→    
   615→    Trade two correlated stocks - short the outperformer, long the underperformer.
   616→    """
   617→    
   618→    def __init__(self, ticker_a: str, ticker_b: str, lookback: int = 60):
   619→        super().__init__(
   620→            name=f"Pairs_{ticker_a}_{ticker_b}",
   621→            parameters={'ticker_a': ticker_a, 'ticker_b': ticker_b, 'lookback': lookback}
   622→        )
   623→        self.ticker_a = ticker_a
   624→        self.ticker_b = ticker_b
   625→        self.lookback = lookback
   626→    
   627→    def generate_signals(self, data: pd.DataFrame) -> pd.DataFrame:
   628→        """
   629→        Generate signals for pairs trading.
   630→        
   631→        data should have columns: ticker_a_close, ticker_b_close
   632→        """
   633→        df = data.copy()
   634→        
   635→        # Calculate normalized prices
   636→        price_a = df[self.ticker_a].fillna(method='ffill')
   637→        price_b = df[self.ticker_b].fillna(method='ffill')
   638→        
   639→        # Calculate price ratios
   640→        ratio = price_a / price_b
   641→        
   642→        # Calculate rolling mean and std of ratio
   643→        ratio_mean = ratio.rolling(window=self.lookback).mean()
   644→        ratio_std = ratio.rolling(window=self.lookback).std()
   645→        
   646→        # Z-score of ratio
   647→        z_score = (ratio - ratio_mean) / ratio_std
   648→        
   649→        # Generate signals
   650→        df['signal'] = 0
   651→        # When ratio is high (A expensive vs B), short A and long B
   652→        df.loc[z_score > 1.5, 'signal'] = -1
   653→        # When ratio is low (A cheap vs B), long A and short B
   654→        df.loc[z_score < -1.5, 'signal'] = 1
   655→        
   656→        df.loc[:self.lookback, 'signal'] = 0
   657→        
   658→        return df[['signal']]
   659→```

   663→### 7. `src/trading_strategies/strategies/stocks/momentum.py`
   664→
   665→```python
   666→"""Momentum strategy from paper (Section 3.1)."""
   667→
   668→import pandas as pd
   669→import numpy as np
   670→from ..base import BaseStrategy
   671→
   672→class MomentumStrategy(BaseStrategy):
   673→    """
   674→    Strategy: Price Momentum (from paper Section 3.1)
   675→    
   676→    Buy stocks with positive recent price momentum.
   677→    Sell stocks with negative momentum.
   678→    """
   679→    
   680→    def __init__(self, lookback: int = 252, percentile_threshold: float = 0.5):
   681→        super().__init__(
   682→            name=f"Momentum_{lookback}",
   683→            parameters={'lookback': lookback, 'percentile_threshold': percentile_threshold}
   684→        )
   685→        self.lookback = lookback
   686→        self.percentile_threshold = percentile_threshold
   687→    
   688→    def generate_signals(self, data: pd.DataFrame) -> pd.DataFrame:
   689→        """
   690→        Generate signals based on historical returns (momentum).
   691→        
   692→        Momentum = (P_today / P_lookback_ago) - 1
   693→        
   694→        Buy when momentum is positive and strong.
   695→        Sell when momentum is negative.
   696→        """
   697→        df = data.copy()
   698→        
   699→        # Calculate momentum (returns over lookback period)
   700→        df['momentum'] = (df['close'] / df['close'].shift(self.lookback) - 1)
   701→        
   702→        # Standardize momentum
   703→        df['momentum_zscore'] = (df['momentum'] - df['momentum'].rolling(60).mean()) / df['momentum'].rolling(60).std()
   704→        
   705→        # Generate signals
   706→        df['signal'] = 0
   707→        df.loc[df['momentum'] > 0, 'signal'] = 1      # Positive momentum: BUY
   708→        df.loc[df['momentum'] < 0, 'signal'] = -1     # Negative momentum: SELL
   709→        
   710→        # Neutral until we have enough data
   711→        df.loc[:self.lookback, 'signal'] = 0
   712→        
   713→        return df[['signal']]
   714→```
